local function gradient(text, startColor, endColor)
	local chars = {}
	for uchar in text:gmatch("[%z\1-\127\194-\244][\128-\191]*") do
		table.insert(chars, uchar)
	end
	local length = #chars
	local result = ""
	for i = 1, length do
		local t = (i - 1) / math.max(length - 1, 1)
		local r = startColor.R + (endColor.R - startColor.R) * t
		local g = startColor.G + (endColor.G - startColor.G) * t
		local b = startColor.B + (endColor.B - startColor.B) * t
		result = result .. string.format('<font color="rgb(%d,%d,%d)">%s</font>',
			math.floor(r * 255), math.floor(g * 255), math.floor(b * 255), chars[i])
	end
	return result
end

local function hasShieldProtection(player)
    if not player or not player.Character then return false end
    local humanoid = player.Character:FindFirstChild("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return false end
    for _, desc in pairs(player.Character:GetDescendants()) do
        if desc:IsA("ForceField") or desc.Name:lower():find("shield") then return true end
    end
    return false
end

local function hasForceField(player)
    local char = player.Character
    if not char then return false end
    if char:FindFirstChildOfClass("ForceField") then return true end
    for _, child in ipairs(char:GetDescendants()) do
        local name = child.Name:lower()
        if name:find("shield", 1, true) or name:find("spawn", 1, true) or name:find("protect", 1, true) then
            return true
        end
    end
    return false
end

local cornerRadius = 22
local glowLayers = {}
local layerConfigs = {}
local maxOffset = 40
local numLayers = 8
for i = 1, numLayers do
	local offset = 4 + (i - 1) * ((maxOffset - 4) / (numLayers - 1))
	local trans = 0.65 + (i - 1) * (0.98 - 0.65) / (numLayers - 1)
	local radius = cornerRadius + (i - 1) * 0.5
	table.insert(layerConfigs, { offset = offset, trans = trans, radius = radius })

end

local function refreshMultiDropdown(dropdown, newList)
    if not dropdown then return end
    pcall(function() dropdown.Values = newList end)
    pcall(function() dropdown.Value = {} end)
    pcall(function() dropdown:Set({ Values = newList, Value = {} }) end)
    pcall(function() dropdown:SetValues(newList) end)
    pcall(function() dropdown:SetValue({}) end)
    pcall(function() dropdown:Refresh(newList, {}) end)
end

local function refreshDropdown(dropdown, newList, defaultValue)
    if not dropdown then return end
    pcall(function() dropdown.Values = newList end)
    pcall(function() dropdown.Value = defaultValue end)
    pcall(function() dropdown:Set({ Values = newList, Value = defaultValue }) end)
    pcall(function() dropdown:SetValues(newList) end)
    pcall(function() dropdown:SetValue(defaultValue) end)
    pcall(function() dropdown:Refresh(newList, defaultValue) end)
end

do
    local nametagEnabled = false
    local LocalHead = nil
    local playerConnections = {}

    local colorSequence = {
        Color3.fromRGB(173, 216, 230),
        Color3.fromRGB(0, 255, 255),
        Color3.fromRGB(0, 120, 255),
        Color3.fromRGB(255, 255, 255)
    }
    local colorT = 0

    local function getGradientColor()
        colorT = colorT + 0.015
        if colorT >= #colorSequence then
            colorT = 0
        end
        local i = math.floor(colorT) + 1
        local j = i % #colorSequence + 1
        local t = colorT - math.floor(colorT)
        return colorSequence[i]:Lerp(colorSequence[j], t)
    end

    local function updateNametag(player, label)
        if not nametagEnabled then return end
        local char = player.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local head = char:FindFirstChild("Head")
        if hum and head and hum.Health > 0 and LocalHead then
            local dist = (LocalHead.Position - head.Position).Magnitude
            label.Text = string.format("%s\n血量: %d/%d\n距离: %.1fm", player.Name, math.floor(hum.Health), math.floor(hum.MaxHealth), dist)
            label.TextColor3 = getGradientColor()
            label.Visible = true
        else
            label.Visible = false
        end
    end

    local function createNametag(player)
        if player == LocalPlayer or not nametagEnabled then return end
        playerConnections[player] = {}

        local function setup(character)
            local head = character:WaitForChild("Head")
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "TestNametag"
            billboard.Adornee = head
            billboard.Size = UDim2.new(0, 200, 0, 80)
            billboard.StudsOffset = Vector3.new(0, 3, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = head

            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.Font = Enum.Font.GothamBold
            label.TextSize = 9
            label.TextStrokeColor3 = Color3.new(0, 0, 0)
            label.TextStrokeTransparency = 0.2
            label.BackgroundTransparency = 1
            label.TextYAlignment = Enum.TextYAlignment.Top
            label.Parent = billboard

            local conn = RunService.Heartbeat:Connect(function()
                if character.Parent then
                    updateNametag(player, label)
                else
                    conn:Disconnect()
                    billboard:Destroy()
                end
            end)
            table.insert(playerConnections[player], conn)
        end

        if player.Character then
            setup(player.Character)
        end
        local charAdd = player.CharacterAdded:Connect(setup)
        table.insert(playerConnections[player], charAdd)
    end

    local function removeNametag(player)
        if playerConnections[player] then
            for _, c in ipairs(playerConnections[player]) do
                c:Disconnect()
            end
            playerConnections[player] = nil
        end
        if player.Character then
            local head = player.Character:FindFirstChild("Head")
            if head then
                local nt = head:FindFirstChild("TestNametag")
                if nt then nt:Destroy() end
            end
        end
    end

    local function init()
        LocalHead = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head")
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then createNametag(p) end
        end
        Players.PlayerAdded:Connect(createNametag)
        Players.PlayerRemoving:Connect(removeNametag)
        LocalPlayer.CharacterAdded:Connect(function(c)
            LocalHead = c:WaitForChild("Head")
        end)
    end

    local function clear()
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then removeNametag(p) end
        end
        colorT = 0
    end

    DrawTab:Toggle({
        Title = L.ESP,
        Default = false,
        Callback = function(Value)
            nametagEnabled = Value
            if Value then init() else clear() end
        end
    })
end

end

local function performRPGBomb(rounds)
    rounds = rounds or 1
    if not inv.getFromName("RPG") then
        sig.InvokeServer("attemptPurchase", "RPG")
        task.wait(0.3)
    end
    local equip = getEquippedItemCompat()
    if not equip or equip.name ~= "RPG" then
        local rpg = inv.getFromName("RPG")
        if rpg then
            sig.FireServer("equip", rpg.guid)
            task.wait(0.1)
            equip = getEquippedItemCompat()
        end
    end
    if not equip or equip.name ~= "RPG" then return end
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        sig.FireServer("replicateProjectiles", equip.guid, { { "AmmoGuid", hrp.CFrame } }, "semi")
    end
    for round = 1, rounds do
        bombAllPlayersOnce()
        if round < rounds then task.wait(0.5) end
    end
end

local function realFireRPGAura()
    local equip = getEquippedItemCompat()
    if not equip or equip.name ~= "RPG" then return nil end
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local targetCF = CFrame.new(hrp.Position + Vector3.new(0, 100, 0))
    local newGuid = guid()
    pcall(function()
        sig.FireServer("replicateProjectiles", equip.guid, { { newGuid, targetCF } }, "semi")
    end)
    task.wait(0.1)
    pcall(function()
        sig.InvokeServer("attemptPurchaseAmmo", "RPG")
        sig.FireServer("reload", equip.guid)
    end)
    task.wait(0.2)
    return newGuid
end

local function bombAuraOnce()
    if not currentAuraAmmoGuid then return end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer
            and not table.find(NinjaWhitelist, player.UserId)
            and not (NinjaBlacklistEnabled and not table.find(NinjaBlacklistTargets, player.Name))
            and player.Character
            and player.Character:FindFirstChild("HumanoidRootPart")
            and not player.Character:FindFirstChildOfClass("ForceField")
        then
            local rootPart = player.Character.HumanoidRootPart
            sig.FireServer("rocketHit", currentAuraAmmoGuid, "explosionGUID", rootPart.Position)
        end
    end
end

local function rpgAuraMainLoop()
    local lastRefresh = tick()
    while RPGAuraEnabled do
        if tick() - lastRefresh >= auraRefreshInterval then
            local newGuid = realFireRPGAura()
            if newGuid then
                currentAuraAmmoGuid = newGuid
                lastRefresh = tick()
            end
        end
        bombAuraOnce()
        task.wait(0.05)
    end
end

local function gkThrow()

local function gkCreateTrace(targetPos)
    local char = LocalPlayer.Character
    if not (char and char.PrimaryPart) then return end
    local S = char.PrimaryPart.Position + Vector3.new(math.random(-15,15), math.random(-15,15), math.random(-15,15))
    local mag = (targetPos - S).Magnitude
    local P = Instance.new("Part")
    P.Name = "Trace"
    P.Anchored = true
    P.CanCollide = false
    P.CastShadow = false
    P.Material = Enum.Material.Neon
    P.Color = Color3.fromHSV(tick() % 1, 0.8, 1)
    P.Size = Vector3.new(0.15, 0.15, mag)
    P.CFrame = CFrame.lookAt(S, targetPos) * CFrame.new(0, 0, -mag/2)
    P.Parent = workspace
    game:GetService("TweenService"):Create(P, TweenInfo.new(0.3), { Transparency = 1, Size = Vector3.new(0,0,mag) }):Play()
    game:GetService("Debris"):AddItem(P, 0.3)
end

RunService.Heartbeat:Connect(function()
    if not gkAuraBlade then return end
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local target, dist = nil, math.huge
    for _, v in pairs(Players:GetPlayers()) do
        if not table.find(NinjaWhitelist, v.UserId)
            and not (NinjaBlacklistEnabled and not table.find(NinjaBlacklistTargets, v.Name))
        then
            local isTarget = table.find(gkTargetPlayers, "ALL") or table.find(gkTargetPlayers, v.Name)
            if isTarget and v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") and v.Character.Humanoid.Health > 0 and not v.Character:FindFirstChildOfClass("ForceField") and not LocalPlayer:IsFriendsWith(v.UserId) then
                local pos = v.Character.HumanoidRootPart.Position
                local mag = (myRoot.Position - pos).Magnitude
                if mag < dist then
                    dist = mag
                    target = v.Character.Head
                end
            end
        end
    end
    if target then
        local targetPos = target.Position
        local currentTime = tick()
        if currentTime - gkLastAttack >= 0 then
            if getgenv().fbsx == "Gun Kill" then
                local item = dvv.load("v3item").GetEquipped(LocalPlayer)
                if item and item.type == "Gun" then
                    gkLastAttack = currentTime
                    if item.ammoManager and item.ammoManager.ammo > 0 then
                        local g = guid()
                        gkCreateTrace(targetPos)
                        pcall(function()
                            sig.FireServer("replicateProjectiles", item.guid, { { g, target.CFrame } }, item.firemode)
                            sig.FireServer("projectileHit", g, "player", {
                                hitSize = target.Size, hitPart = target, pos = targetPos,
                                hitPlayerId = Players:GetPlayerFromCharacter(target.Parent).UserId
                            })
                        end)
                        item.ammoManager.ammo = item.ammoManager.ammo - 1
                    else
                        pcall(function()
                            sig.InvokeServer("attemptPurchaseAmmo", item.name)
                            sig.FireServer("reload", item.guid)
                        end)
                    end
                end
            elseif getgenv().bladeid then
                gkLastAttack = currentTime
                gkCreateTrace(targetPos)
                pcall(function() sig.InvokeServer("hitSticky", getgenv().bladeid, target, target.CFrame, target.CFrame) end)
            end
        end
    end
end)

local function hasShield(char)
    if not char then return false end
    if char:FindFirstChildOfClass("ForceField") then return true end
    for _, d in pairs(char:GetDescendants()) do
        if d.Name:lower():find("shield") then return true end
    end
    return false
end

local function isFriend(plr)
    return LocalPlayer:IsFriendsWith(plr.UserId)
end

local function getBulletTarget()
    local target = nil
    local minDist = math.huge
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer
            and not table.find(NinjaWhitelist, plr.UserId)
            and not (NinjaBlacklistEnabled and not table.find(NinjaBlacklistTargets, plr.Name))
            and plr.Character
        then
            local char = plr.Character
            local part = char:FindFirstChild(BulletConfig.ShootPart)
            local hum = char:FindFirstChildOfClass("Humanoid")
            if part and hum and hum.Health > 0 then
                if not (BulletConfig.CheckShield and hasShield(char)) and not (BulletConfig.CheckFriend and isFriend(plr)) then
                    local dist = (part.Position - myRoot.Position).Magnitude
                    if dist < minDist then
                        minDist = dist
                        target = part
                    end
                end
            end
        end
    end
    return target
end

        if bulletTrackingHook then hookmetamethod(game, "__namecall", bulletTrackingHook) bulletTrackingHook = nil end
        if Value then
            bulletTrackingHook = hookmetamethod(game, "__namecall", function(self, ...)
                local m = getnamecallmethod()
                local args = {...}
                if m == "Raycast" and not checkcaller() then
                    local origin = args[1] or Camera.CFrame.Position
                    local tar = getBulletTarget()
                    if tar then
                        return {
                            Instance = tar,
                            Position = tar.Position,
                            Normal = (origin - tar.Position).Unit,
                            Material = Enum.Material.Plastic,
                            Distance = (tar.Position - origin).Magnitude
                        }
                    end
                end
                return bulletTrackingHook(self, ...)
            end)
        end
    end
})

local itemMap = {}
local function updateCache()
    local itemPickup = Workspace.Game.Entities:FindFirstChild("ItemPickup")
    if not itemPickup then itemMap = {} return end
    itemMap = {}
    for _, model in pairs(itemPickup:GetChildren()) do
        for _, v in pairs(model:GetChildren()) do
            if v:IsA("MeshPart") or v:IsA("Part") then
                local prompt = v:FindFirstChildOfClass("ProximityPrompt")
                if prompt and prompt.ObjectText then
                    itemMap[prompt.ObjectText] = { part = v, prompt = prompt }
                end
            end
        end
    end
end
updateCache()

    itemPickupFolder.ChildAdded:Connect(updateCache)
    itemPickupFolder.ChildRemoved:Connect(updateCache)

local itemFailCounts = {}

local function Autoitem(itemName)
    local itemData = itemMap[itemName]
    if not itemData then
        itemFailCounts[itemName] = (itemFailCounts[itemName] or 0) + 1
        return false
    end
    local char = LocalPlayer.Character
    if not char then
        itemFailCounts[itemName] = (itemFailCounts[itemName] or 0) + 1
        return false
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then
        itemFailCounts[itemName] = (itemFailCounts[itemName] or 0) + 1
        return false
    end
    root.CFrame = itemData.part.CFrame
    itemData.prompt.RequiresLineOfSight = false
    itemData.prompt.HoldDuration = 0
    fireproximityprompt(itemData.prompt)
    itemFailCounts[itemName] = 0
    return true
end

local function runATMPhase()
    local atms = workspace:FindFirstChild("ATMs")
    if not atms then return end
    local root = getRoot()
    if not root then return end
    busy = true
    local originalCFrame = root.CFrame
    local currentTarget = nil
    local stop = false
    local lockConn = nil
    local canSearch = true

    local function clearLock()
        if lockConn then lockConn:Disconnect() lockConn = nil end
    end

    local moveThread = task.spawn(function()
        while not stop do
            if canSearch then
                local nearestATM, nearestDist = nil, math.huge
                for _, atm in ipairs(atms:GetChildren()) do
                    if atm:IsA("Model") and (atm:GetAttribute("health") or 0) ~= 0 then
                        local main = atm:FindFirstChild("Main")
                        if main and main:IsA("BasePart") then
                            local dist = (root.Position - main.Position).Magnitude
                            if dist < nearestDist then nearestDist = dist nearestATM = atm end
                        end
                    end
                end
                if not nearestATM then stop = true clearLock() break end
                currentTarget = nearestATM
                local main = nearestATM:FindFirstChild("Main")
                local lockCF = CFrame.new(main.Position - Vector3.new(0, 4, 0)) * CFrame.Angles(math.rad(90), 0, 0)
                root.CFrame = lockCF
                clearLock()
                lockConn = RunService.Heartbeat:Connect(function()
                    if root and root.Parent then root.CFrame = lockCF root.Velocity = Vector3.zero root.RotVelocity = Vector3.zero end
                end)
                canSearch = false
                task.spawn(function()
                    task.wait(0.4)
                    if currentTarget and currentTarget.Parent then currentTarget:SetAttribute("health", 0) collectNearbyCash() end
                    task.wait(1)
                    currentTarget = nil
                    canSearch = true
                end)
            end
            task.wait(0.1)
        end
        clearLock()
    end)

    repeat task.wait(0.1) until stop
    if root then root.CFrame = originalCFrame end
    busy = false
end

local function ensureItem(itemName, buyLocation)
    if not InvokeServer or not FireServer then return end
    local items = require(ReplicatedStorage.devv).load("v3item").inventory.items
    for _, v in pairs(items) do
        if v.name == itemName then return v.guid end
    end
    local root = getRoot()
    if root then
        root.CFrame = buyLocation
        task.wait(0.5)
        InvokeServer("attemptPurchase", itemName)
        task.wait(0.5)
    end
    items = require(ReplicatedStorage.devv).load("v3item").inventory.items
    for _, v in pairs(items) do
        if v.name == itemName then return v.guid end
    end
    return nil
end

local function tryBankHeist()
    if not FireServer or not InvokeServer then return false end
    busy = true
    local bank = Workspace:FindFirstChild("BankRobbery")
    if not bank then busy = false return false end
    local cashFolder = bank:FindFirstChild("BankCash") and bank.BankCash:FindFirstChild("Cash")
    if not cashFolder or #cashFolder:GetChildren() == 0 then busy = false return false end
    local fragGuid = ensureItem("Frag", grenadeBuyLocation)
    if not fragGuid then busy = false return false end
    local root = getRoot()
    if not root then busy = false return false end
    root.CFrame = bombThrowLocation
    task.wait(0.3)
    FireServer("equip", fragGuid)
    task.wait(0.1)
    local direction = (bombTargetPosition - root.Position).Unit
    FireServer("throwItem", fragGuid, direction, bombTargetPosition)
    FireServer("removeItem", fragGuid)
    task.wait(1.5)
    bank = Workspace:FindFirstChild("BankRobbery")
    if not bank then busy = false return false end
    cashFolder = bank:FindFirstChild("BankCash") and bank.BankCash:FindFirstChild("Cash")
    local pallet = bank:FindFirstChild("BankCash") and bank.BankCash:FindFirstChild("Pallet")
    local promptPart = bank:FindFirstChild("BankCash") and bank.BankCash:FindFirstChild("Main")
    local prompt = promptPart and promptPart:FindFirstChild("Attachment") and promptPart.Attachment:FindFirstChild("ProximityPrompt")
    if not (cashFolder and pallet and prompt) then busy = false return false end
    root.CFrame = afterExplosionWaitLocation
    local lockPos = (afterExplosionWaitLocation * CFrame.new(0, 0, -4)).Position
    local lockCF = CFrame.new(lockPos) * CFrame.Angles(math.rad(90), 0, 0)
    local endTime = tick() + 4
    while tick() < endTime do
        root = getRoot()
        if root then root.CFrame = lockCF root.Velocity = Vector3.zero root.RotVelocity = Vector3.zero end
        task.wait(0.2)
    end
    local collectCF = CFrame.new((pallet.CFrame * CFrame.new(0, -2, 0)).Position) * CFrame.Angles(math.rad(90), 0, 0)
    root = getRoot()
    if root then root.CFrame = collectCF end
    local cashConnection
    cashConnection = RunService.Heartbeat:Connect(function()
        if not cashFolder or not cashFolder.Parent or #cashFolder:GetChildren() == 0 then
            if cashConnection then cashConnection:Disconnect() end
            return
        end
        collectNearbyCash()
        pcall(function() fireproximityprompt(prompt) end)
    end)
    while bank and cashFolder and prompt.Enabled and #cashFolder:GetChildren() > 0 do
        root = getRoot()
        if not root then break end
        root.CFrame = collectCF
        root.Velocity = Vector3.zero
        root.RotVelocity = Vector3.zero
        task.wait(0.2)
    end
    if cashConnection then cashConnection:Disconnect() end
    root = getRoot()
    if root then root.CFrame = idleLocation root.Velocity = Vector3.zero root.RotVelocity = Vector3.zero end
    busy = false
    return true
end
